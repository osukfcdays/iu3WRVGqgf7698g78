ClientFighter.GetCameraSway = function(arg, arg2)
	local firstPerson = CameraController.CameraState.States.FirstPerson
	if CameraController:GetPublicState() ~= firstPerson then
		return Vector2.zero
	end
	if arg.IsLocalPlayer then
		if ControlsController.CurrentControls == "Gamepad" then
			return CameraController:GetGamepadRotationInput(arg2) * 5
		end
		return UserInputService:GetMouseDelta() * CameraController:GetCameraSensitivity() * -0.1
	end
	local v = arg._sway_camera_rotations[1]
	local v2 = arg._sway_camera_rotations[2]
	if not v or not v2 then
		return Vector2.zero
	end
	local swayCameraRotationUpdateTime = arg._sway_camera_rotation_update_time
	math.clamp((tick() - swayCameraRotationUpdateTime) / SWAY_UPDATE_INTERVAL, 0, 1)
	local n2 = v - v2
	return arg2 and Vector2.new(n2.Y, n2.X) or n2
end

ClientFighter.EquipItem = function(arg, arg2, arg3)
	assert(not arg2 or typeof(arg2) == "number", "Argument 1 invalid, expected a number or nil")
	local flag = arg:IsEmoting()
	if not flag then
		local equipCooldownFromEmoting = arg._equip_cooldown_from_emoting
		flag = tick() < equipCooldownFromEmoting
	end
	if flag or EmoteController:IsEquippingItemsDisabled() then
		return
	end
	local ObjectID = arg2 and arg.Items[arg2]
	if arg2 and not ObjectID then
		return
	end
	ObjectID = ObjectID and ObjectID:Get("ObjectID") or nil
	if ObjectID == arg:Get("LocalEquippedItemID") then
		return
	end
	if not arg3 and arg:IsEquippingLockedByQuickAttacks(ObjectID) then
		return
	end
	if arg.Entity and arg.Entity:Get("IsFrozen") then
		return
	end
	arg:SetReplicate("LocalEquippedItemID", ObjectID)
	if not ReplicatedStorage.Remotes.Replication.Fighter.EquipItemFeedback:InvokeServer(ObjectID) and arg:Get("LocalEquippedItemID") ~= arg:Get("EquippedItemID") then
		arg:SetReplicate("LocalEquippedItemID", arg:Get("EquippedItemID"))
	end
end

ClientFighter.EquipIncrement = function(arg, arg2)
	assert(typeof(arg2) == "number", "Argument 1 invalid, expected a number")
	local LocalEquippedItemID = arg:Get("LocalEquippedItemID")
	local n2
	if LocalEquippedItemID then
		n2 = nil
		for k, item in pairs(arg.Items) do
			if item:Get("ObjectID") == LocalEquippedItemID then
				n2 = (k + arg2 - 1) % #arg.Items + 1
				break
			else
				n2 = nil
			end
		end
	else
		n2 = arg2 < 0 and #arg.Items or 1
	end
	if n2 then
		arg:EquipItem(n2)
	end
end

ClientFighter.Input = function(arg, arg2)
	if arg:AreItemsLocked() and not string.find(arg2, "Inspect") then
		return
	end
	if arg:IsEmoting() and #arg2 >= 5 and string.sub(arg2, 1, 5) == "Start" then
		arg.Entity:CancelEmote()
		ReplicatedStorage.Remotes.Replication.Fighter.CancelEmoteFromClient:FireServer()
		arg._inputs_locked_from_emotes = tick() + 0.2
		return
	end

	for k in pairs(equippedItems) do

	end
end
